```dockerfile
FROM python:3.12-slim

# System deps for psycopg binary
RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq-dev \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Poetry install
ENV POETRY_VERSION=1.8.0 \
    POETRY_VIRTUALENVS_CREATE=false \
    POETRY_NO_INTERACTION=1 \
    POETRY_CACHE_DIR=/tmp/poetry_cache

RUN pip install --no-cache-dir "poetry==${POETRY_VERSION}"

WORKDIR /app

# Install dependencies first (layer cache)
COPY pyproject.toml poetry.lock* ./
RUN poetry install --only main --no-root && rm -rf ${POETRY_CACHE_DIR}

# Copy application source
COPY . .

EXPOSE 8001

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8001", "--reload"]
```

Read the current /Users/sanchayjain/Desktop/ERM/docker-compose.yml in full. Add the vdr-agent service entry after the ingestion-service block. Do NOT modify any existing service entries.

Add this block:

```yaml
  # ─── VDR Agent Service ────────────────────────────────────────────────────
  vdr-agent:
    build:
      context: ./vdr-agent
      dockerfile: Dockerfile
    container_name: erm-vdr-agent
    ports:
      - "8001:8001"
    extra_hosts:
      - "host.docker.internal:host-gateway"
    env_file:
      - ./vdr-agent/.env.local
    environment:
      - VDR_AGENT_DB_HOST=postgres
      - VDR_AGENT_ENV=local
    volumes:
      - ./vdr-agent/app:/app/app
      - ./vdr-agent/main.py:/app/main.py
    command: ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8001", "--reload"]
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - erm-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8001/vdr-agent/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
```

IMPORTANT: Check the exact service name for PostgreSQL in docker-compose.yml before writing `depends_on`. If the postgres service is named differently (e.g., `db`, `postgresql`), use that name. If no postgres healthcheck condition is available, use `depends_on: [postgres]` without condition.

Also check whether ingestion-service uses port 8001 on the host — if so, the existing comment at the top says ingestion-service maps 8001:8000. vdr-agent needs 8001:8001, which conflicts. In that case, change vdr-agent host port to 8004:8001 and update the healthcheck URL accordingly. The container port stays 8001.
